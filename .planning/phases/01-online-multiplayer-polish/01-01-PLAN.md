```
public class RematchManager : MonoBehaviour

    public static RematchManager Instance { get; private set; }

    // Configuration
    public float RematchTimerDuration = 30f;

    // State
    private Dictionary<Player, RematchStatus> _playerStatuses;
    private float _timerStartTime;
    private bool _timerActive;

    // Events
    public event Action<Player, RematchStatus> OnPlayerStatusChanged;
    public event Action<float> OnTimerStarted;  // passes duration
    public event Action OnTimerExpired;
    public event Action<List<Player>> OnRematchConfirmed;  // passes confirmed players
    public event Action OnRematchCancelled;  // not enough players

    // Methods
    public void Initialize(int playerCount)  // Reset state for new end-game
    public void SetPlayerStatus(Player player, RematchStatus status)
    public RematchStatus GetPlayerStatus(Player player)
    public void StartTimer()  // Called when first player confirms
    public float GetTimeRemaining()
    public bool IsTimerActive => _timerActive
    public List<Player> GetConfirmedPlayers()
    public void HandleTimerExpired()  // Check if enough players, fire appropriate event
    public void Reset()  // Clean up for new game
}
```

Key behaviors:
- When first player sets Confirmed, start timer and fire OnTimerStarted
- When player sets Declined (or leaves), fire OnPlayerStatusChanged immediately
- Toggle support: player can go Confirmed → Pending → Confirmed
- In Update(), check timer expiration and call HandleTimerExpired()
- HandleTimerExpired: if 2+ confirmed → OnRematchConfirmed, else OnRematchCancelled

```csharp
public struct NetworkRematchStatus : INetworkSerializable
{
    public byte PlayerIndex;        // 0=Yellow, 1=Blue, 2=Purple, 3=Pink
    public byte Status;             // 0=Pending, 1=Confirmed, 2=Declined
    public float TimerStartTime;    // Server time when timer started (0 if not started)
    public float TimerDuration;     // Duration in seconds

    public Player GetPlayer() => (Player)PlayerIndex;
    public RematchStatus GetStatus() => (RematchStatus)Status;

    public void NetworkSerialize<T>(BufferSerializer<T> serializer) where T : IReaderWriter
    {
        serializer.SerializeValue(ref PlayerIndex);
        serializer.SerializeValue(ref Status);
        serializer.SerializeValue(ref TimerStartTime);
        serializer.SerializeValue(ref TimerDuration);
    }
}
```

**In NetworkGameBridge.cs**, update RPCs:

```csharp
// Add event
public event Action<NetworkRematchStatus> OnRematchStatusReceived;

// Update ServerRpc to use new struct
[ServerRpc(RequireOwnership = false)]
public void RequestRematchStatusServerRpc(NetworkRematchStatus status, ServerRpcParams rpcParams = default)
{
    // Host validates sender matches player
    Player senderPlayer = GetPlayerFromClientId(rpcParams.Receive.SenderClientId);
    if ((int)senderPlayer != status.PlayerIndex)
    {
        Debug.LogWarning($"[NetworkGameBridge] Rematch status mismatch: sender is {senderPlayer} but status is for {status.GetPlayer()}");
        return;
    }

    // If this is first Confirmed and timer not started, set timer fields
    if (status.GetStatus() == RematchStatus.Confirmed && status.TimerStartTime == 0)
    {
        status.TimerStartTime = Time.time;
        status.TimerDuration = RematchManager.Instance?.RematchTimerDuration ?? 30f;
    }

    BroadcastRematchStatusClientRpc(status);
}

[ClientRpc]
private void BroadcastRematchStatusClientRpc(NetworkRematchStatus status)
{
    Debug.Log($"[NetworkGameBridge] Rematch status: {status.GetPlayer()} = {status.GetStatus()}");
    OnRematchStatusReceived?.Invoke(status);
}
```

```csharp
NetworkGameBridge.Instance.OnRematchStatusReceived += OnNetworkRematchStatusReceived;

private void OnNetworkRematchStatusReceived(NetworkRematchStatus status)
{
    // Update local RematchManager with received status
    if (RematchManager.Instance != null)
    {
        RematchManager.Instance.SetPlayerStatus(status.GetPlayer(), status.GetStatus());

        // If timer info included and timer not yet active locally, sync it
        if (status.TimerStartTime > 0 && !RematchManager.Instance.IsTimerActive)
        {
            RematchManager.Instance.SyncTimer(status.TimerStartTime, status.TimerDuration);
        }
    }
}
```
